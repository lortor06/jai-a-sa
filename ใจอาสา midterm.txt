          if (log.activityId === actId) {
            updates['logs/' + logId] = null;
          }
        });
        Object.entries(evaluationsDB).forEach(([evalId, ev]) => {
          if (ev.activityId === actId) {
            updates['evaluations/' + evalId] = null;
          }
        });
        if (Object.keys(updates).length > 0) {
          update(ref(db), updates);
        }
        alert('ลบกิจกรรมเรียบร้อยแล้ว');
      }).catch(err => alert('เกิดข้อผิดพลาด: ' + err.message));

  function renderMembers() {

    const search = (document.getElementById('searchMemberInput')?.value || '').toLowerCase();
    const studentEntries = allStudents.filter(([id, student]) => {
      return (student.name?.toLowerCase().includes(search) || student.studentId?.toLowerCase().includes(search));
    });

    if (studentEntries.length === 0) {
      document.getElementById('adminMemberList').innerHTML = '<p style="text-align:center;">ไม่พบข้อมูลนักศึกษาที่ตรงกับการค้นหา</p>';

        .filter(log => log.studentId === student.studentId && log.status === 'approved' && activitiesDB[log.activityId])

  window.exportMembersToCSV = function() {
    const allStudents = Object.entries(usersDB).filter(([id, user]) => user.role === 'user');
    if (allStudents.length === 0) return alert('ไม่มีข้อมูลนักศึกษาให้ส่งออก');
    
    let csvContent = "\uFEFFลำดับ,รหัสนักศึกษา,ชื่อ-นามสกุล,อีเมล,ชั่วโมงจิตอาสาอนุมัติแล้ว (ชม.)\n";
    
    allStudents.forEach(([uid, student], index) => {
      const approvedHours = Object.values(logsDB)
        .filter(log => log.studentId === student.studentId && log.status === 'approved' && activitiesDB[log.activityId])
        .reduce((sum, log) => sum + (parseFloat(log.hours) || 0), 0);
      csvContent += `"${index + 1}","${student.studentId}","${student.name}","${student.email}","${approvedHours}"\n`;
    });
    
    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
    const link = document.createElement("a");
    link.href = URL.createObjectURL(blob);
    link.setAttribute("download", `รายงานสรุปชั่วโมงจิตอาสา_${new Date().toLocaleDateString('th-TH').replace(/\//g, '-')}.csv`);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  }

  function renderUserActivities() { 
    const actArray = Object.entries(activitiesDB).sort((a,b) => b[1].timestamp - a[1].timestamp);
    if(actArray.length === 0) {
      document.getElementById('userActivityList').innerHTML = '<p style="text-align:center;">ยังไม่มีกิจกรรมเปิดให้เข้าร่วม</p>';
      return;
    }

    const today = new Date();
    today.setHours(0,0,0,0);

    document.getElementById('userActivityList').innerHTML = actArray.map(([id, a]) => {
      const actDate = new Date(a.date);
      actDate.setHours(0,0,0,0);
      
      const diffTime = today.getTime() - actDate.getTime();
      const diffDays = Math.floor(diffTime / (1000 * 60 * 60 * 24));

      const isSubmitted = Object.values(logsDB).some(
        l => l.activityId === id && l.studentId === (currentUser ? currentUser.studentId : '')
      );

      let btnHtml = '';
      let dateNotice = '';

      if (isSubmitted) {
        btnHtml = `<button class="btn small" disabled style="background:#E5E7EB; cursor:not-allowed; border:none;">✅ ส่งแล้ว</button>`;
        dateNotice = `<span style="color:var(--primary); font-size:11.5px; display:block; margin-top:2px;">(ส่งหลักฐานเรียบร้อยแล้ว)</span>`;
      } else if (diffDays < 0) {
        btnHtml = `<button class="btn small" disabled style="background:#E5E7EB; cursor:not-allowed; border:none;">⏳ ยังไม่ถึงวันจัดกิจกรรม</button>`;
        dateNotice = `<span style="color:var(--gold); font-size:11.5px; display:block; margin-top:2px;">(เปิดให้บันทึกในวันที่ ${a.date})</span>`;
      } else if (diffDays > 7) {
        btnHtml = `<button class="btn small danger" disabled style="cursor:not-allowed; opacity:0.6;">⛔ เลยกำหนด 7 วันแล้ว</button>`;
        dateNotice = `<span style="color:var(--rust); font-size:11.5px; display:block; margin-top:2px;">(หมดเขตส่งเมื่อ ${new Date(actDate.getTime() + 7*24*60*60*1000).toLocaleDateString('th-TH')})</span>`;
      } else {
        btnHtml = `<button class="btn small" onclick="joinActivity('${id}')" style="background:var(--primary); color:#fff; border:none;">ส่งภาพหลักฐาน</button>`;
        dateNotice = `<span style="color:var(--primary); font-size:11.5px; display:block; margin-top:2px;">(สามารถส่งหลักฐานได้อีก ${7 - diffDays} วัน)</span>`;
      }

      return `
        <div class="activity-item" style="align-items:flex-start;">
          <div>
            <h4 style="margin:0; color:var(--primary);">${a.titleTh}</h4>
            <div style="font-size:12px; color:var(--text-secondary); margin-top:4px;">
              📍 สถานที่: <strong>${a.location || '-'}</strong><br>
              📅 ${a.date} | ⏰ ${a.timeStart || a.time} - ${a.timeEnd || '?'} น.<br>
              📝 ${a.details || '-'}
              ${dateNotice}
            </div>
          </div>
          <div style="display:flex; flex-direction:column; align-items:flex-end; gap:8px;">
            <span style="color:var(--primary); font-size:14px; font-weight:bold;">⭐ ${a.hours} ชม.</span>
            ${btnHtml}
          </div>
        </div>
      `;
    }).join('');
  }

  window.joinActivity = function(actId) {
    const act = activitiesDB[actId];
    if (!act) return;

    const today = new Date();
    today.setHours(0,0,0,0);
    
    const actDate = new Date(act.date);
    actDate.setHours(0,0,0,0);

    const diffTime = today.getTime() - actDate.getTime();
    const diffDays = Math.floor(diffTime / (1000 * 60 * 60 * 24));

    if (diffDays < 0) {
      return alert(`ยังไม่ถึงวันจัดกิจกรรม! (กิจกรรมนี้กำหนดจัดวันที่ ${act.date})\nคุณสามารถส่งหลักฐานได้ตั้งแต่วันจัดกิจกรรมเป็นต้นไป`);
    }
    if (diffDays > 7) {
      return alert(`หมดเวลาส่งหลักฐานแล้ว! (ต้องส่งภายใน 7 วันหลังวันจัดกิจกรรม)`);
    }

    if(Object.values(logsDB).some(l => l.activityId === actId && l.studentId === currentUser.studentId)) {
      return alert('คุณได้ส่งหลักฐานกิจกรรมนี้ไปเรียบร้อยแล้ว');
    }
    currentJoiningActId = actId;
    document.getElementById('proofFileInput').click();
  }

  window.handleSaveEvaluation = function(e) {
    e.preventDefault();
    const actId = document.getElementById('eval_actId').value;
    const rating = parseInt(document.getElementById('eval_rating').value);
    const comment = document.getElementById('eval_comment').value.trim();
    const act = activitiesDB[actId];

    let existingKey = null;
    for (const [key, ev] of Object.entries(evaluationsDB)) {
      if (ev.studentId === currentUser.studentId && ev.activityId === actId) {
        existingKey = key;
        break;
      }
    }

    const evalData = {
      activityId: actId,
      titleTh: act ? act.titleTh : '',
      studentId: currentUser.studentId,
      studentName: currentUser.name,
      rating: rating,
      comment: comment,
      timestamp: Date.now()
    };

    if (existingKey) {
      update(ref(db, 'evaluations/' + existingKey), evalData).then(() => {
        alert('อัปเดตการประเมินกิจกรรมเรียบร้อยแล้ว!');
        closeEvalModal();
      });
    } else {
      push(ref(db, 'evaluations'), evalData).then(() => {
        alert('ส่งการประเมินกิจกรรมสำเร็จ ขอบคุณครับ!');
        closeEvalModal();
      });
    }
  }

  const KYS_RATES = [1.5, 2.5, 3.0, 3.5, 4.0, 4.5, 5.0, 6.0, 7.0, 8.0, 9.0, 10.0, 11.0, 12.0, 13.0];
  window.handleCalculateLoan = function(e) {
    e.preventDefault();
    const startYear = parseInt(document.getElementById('l_currentYear').value);
    const targetYear = parseInt(document.getElementById('l_targetYear').value);
    const tuition = parseFloat(document.getElementById('l_tuition').value) || 0;
    const includeLiving = document.getElementById('l_includeLiving').checked;
    
    if (targetYear < startYear) return alert('ปีที่จบการศึกษาต้องไม่น้อยกว่าปีที่เริ่มกู้!');

    let cumulativeDebt = (targetYear - startYear + 1) * ((tuition * 2) + (includeLiving ? 36000 : 0));
    document.getElementById('resFutureDebt').textContent = cumulativeDebt.toLocaleString() + ' บาท';

    let remainingPrincipal = cumulativeDebt, repaymentTableRowsHtml = '';
    for (let year = 1; year <= 15; year++) {
      const principalPayment = cumulativeDebt * (KYS_RATES[year - 1] / 100);
      const interestPayment = remainingPrincipal * 0.01;
      const totalYearlyPayment = principalPayment + interestPayment;
      remainingPrincipal -= principalPayment;
      if (remainingPrincipal < 0) remainingPrincipal = 0;
      repaymentTableRowsHtml += `<tr><td>ปีที่ ${year}</td><td>${principalPayment.toLocaleString(undefined,{maximumFractionDigits:0})}</td><td>${interestPayment.toLocaleString(undefined,{maximumFractionDigits:0})}</td><td style="font-weight:700;">${totalYearlyPayment.toLocaleString(undefined,{maximumFractionDigits:0})}</td><td>${remainingPrincipal.toLocaleString(undefined,{maximumFractionDigits:0})}</td></tr>`;
    }
    document.getElementById('repaymentTableBody').innerHTML = repaymentTableRowsHtml;
    document.getElementById('loanResults').style.display = 'block';
  }
